/** 属性値用の簡易エスケープ */
function escapeAttr(str) {
  return String(str)
    .replace(/&/g, "&amp;")
    .replace(/"/g, "&quot;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;");
}

/** id のアイコンを、指定した大きさの <svg> 文字列で返す。aria-label に label を入れる */
export function iconSvg(id, size = 32, label = "") {
  const safeSize = Number.isFinite(Number(size)) && Number(size) > 0 ? Number(size) : 32;
  const safeLabel = escapeAttr(label);
  const raw = Object.prototype.hasOwnProperty.call(ICONS, id) ? ICONS[id] : null;
  if (!raw) {
    // 存在しない id の場合は灰色の四角を返す
    return `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 64 64" width="${safeSize}" height="${safeSize}" role="img" aria-label="${safeLabel}"><rect x="8" y="8" width="48" height="48" rx="8" fill="#7d8694" stroke="#0b2447" stroke-width="4"/></svg>`;
  }
  return raw.replace(
    "<svg ",
    `<svg width="${safeSize}" height="${safeSize}" role="img" aria-label="${safeLabel}" `
  );
}

### 補足（描くのが難しかったアイコン・表からの選択）

- **描くのが難しかったアイコン**:
  - `ab-coord`: 2つの歯車の歯が 24px 表示で潰れず、かつ `viewBox` の端に接して切れないように歯数（8歯と6歯）と座標を調整するのに手間がかかりました。
  - `field-transport`・`ev-scandal`・`gov-nemawashi`: 2要素が重なる構図を 24px でも判別できるよう、前後の重なりや太い輪郭（橋塔・虫眼鏡の柄・耳打ちの2人）のバランスを微調整しました。
- **表の内容からの選択・調整**:
  - `st-mood-1`: 表で「青紫」と指定されていたため、色見本の青（`#1c78e8`）と明確に区別できる青紫（`#5a52d5`）を使っています。
  - `act-local`: `field-diplomacy` や `ev-summit` と握手の絵柄が重ならないよう、「家並みと歩く人」の構図を選びました。
  - `exp-policy`: `act-study`（開いた本）や `gov-draft`（ペンと書類）と区別するため「巻物」を選びました。
  - `field-education`: 日本のゲームらしさと 24px での視認性から「ランドセル」を選びました。
  - `gov-dissolve`: 24px でも一目でわかる「割れた円」の構図を選びました。
